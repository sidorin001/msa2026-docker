cmake_minimum_required(VERSION 3.22)
project(service1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core HttpServer Network Sql Concurrent)
add_executable(service1 src/main.cpp)
target_link_libraries(service1 PRIVATE Qt6::Core Qt6::HttpServer Qt6::Network Qt6::Sql Qt6::Concurrent)
target_compile_options(service1 PRIVATE -Wall -Wextra -Wpedantic)
